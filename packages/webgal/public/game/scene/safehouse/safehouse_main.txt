; ====================================================================
; 安全屋场景（2026-06-09 重构：多房间架构）
;
; 设计：
; - 安全屋有 4 个房间（客厅 / 主卧 / 次卧 / 走廊）
; - 玩家在 safehouse_main.txt 里用 choose 选房间 → setVar:current_room
; - React 组件（SafehouseCharacterPanel）根据 current_room 显示该房间的角色
; - 角色在 characterRoster.ts 里配置 homeRoom
;
; 进入安全屋流程：
;   1. scavenge_main → 玩家点贫民窟 → jumpScene 进安全屋
;   2. safehouse_main 初始化：
;      - 关闭拾荒 UI（地图/时间控制）
;      - 默认进入客厅（setVar:current_room=living_room）
;   3. 玩家选房间 → setVar:current_room → React 卡片更新
;   4. 玩家点"离开"→ changeScene 回 scavenge_main
; ====================================================================

; 关闭拾荒 UI（这些 UI 在安全屋里不显示）
setVar:show_scavenge_time_control=true;
setVar:show_scavenge_map=false;
setVar:scavenge_show_character_list=false;
setVar:scavenge_show_warehouse=false;
; 保留右上角菜单（角色列表/敌人图鉴/物品图鉴 仍可访问）
setVar:scavenge_show_menu=true;

; 初始化：默认在客厅
setVar:current_room=living_room;

; ====================================================================
; 房间切换（2026-06-09 加）
; 玩家用 choose 选要去哪个房间
; 每个 label 只改 current_room，不切场景（保持一个 safehouse_main）
; ====================================================================

:label_room_living
; 客厅：露西通常在这里
setVar:current_room=living_room;
; 不切场景（player 重新 choose）

:label_room_bedroom_main
; 主卧：玩家自己的房间
setVar:current_room=bedroom_main;

:label_room_bedroom_second
; 次卧：暂时空着
setVar:current_room=bedroom_second;

:label_room_corridor
; 走廊：连接各个房间
setVar:current_room=corridor;

:label_rest
; 休息（暂不实现具体逻辑，只跳回地图；后续可在安全屋组件里加 HP 恢复动画）
changeScene:scavenge/scavenge_main.txt;

:label_organize
; 整理仓库：先打开仓库面板，跳回地图后 UI 仍然显示（CSS 不卸载）
setVar:scavenge_show_warehouse=true;
changeScene:scavenge/scavenge_main.txt;

:label_dispatch
; 派遣探索：直接回地图
changeScene:scavenge/scavenge_main.txt;

:label_leave
; 离开安全屋：直接回地图
changeScene:scavenge/scavenge_main.txt;

; 选项设计：
; 1. 去房间（4 个）
; 2. 休息/整理/派遣/离开（保留原来的功能）
; 注：每个 label 都加 ; 注释，WebGAL parser 不解析 ; 开头的行
choose:去客厅:label_room_living|去主卧:label_room_bedroom_main|去次卧:label_room_bedroom_second|去走廊:label_room_corridor|休息:label_rest|整理仓库:label_organize|派遣探索:label_dispatch|离开安全屋:label_leave;
